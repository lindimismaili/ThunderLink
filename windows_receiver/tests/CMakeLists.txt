function(receiver_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ../src ../../common)
    target_link_libraries(${name} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /UNDEBUG)
        target_compile_definitions(${name} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    endif()
    if(WIN32)
        target_link_libraries(${name} PRIVATE ws2_32 bcrypt)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
